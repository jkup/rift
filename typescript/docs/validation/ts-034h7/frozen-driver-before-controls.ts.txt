type Attempt={stage:string;harnessValidated:boolean;status:string;source:string;host:Error|null};
const hostSentinel=new Error("H7 exact host sentinel");
// Local validation orchestration only; this is deliberately not observation transport.
function attempt(mode:string,preflightFault:boolean):Attempt {
 let stage="harness",validated=false,source="",host:Error|null=null;
 const preRealm=fresh(),preSession=createScriptSession({maxSteps:100000,maxCalls:mode==="resource"?1:64});
 function setup(r:Realm):void {
  const fault=r.createBuiltin("fault",0,(_receiver,_args,_context,loc):ValueCompletion=>{
   if(mode==="host")throw hostSentinel;
   if(mode==="unsupported")return unsupportedFailure("runtime.object-to-string","TypeError","literal unsupported",loc);
   return normal();
  });r.defineScriptGlobal("fault",{value:{kind:"object",object:fault},writable:true,configurable:true});
 }
 function classify(v:ValueCompletion):string {
  if(v.type==="normal")return "normal";
  if(v.origin.kind==="resource-limit")return "resource";
  if(v.origin.kind==="unsupported-operation")return "unsupported";
  return "language-throw";
 }
 const testRealm=fresh(),testSession=createScriptSession({maxSteps:100000,maxCalls:mode==="resource"?1:64});
 try {
  load(preRealm,preSession,"attempt/preflight");setup(preRealm);
  const control=preflightFault?'assert.throws(TypeError,function(){fault();});':'assert(true);';
  const pre=run(control,preRealm,preSession,"attempt/preflight/control");
  if(pre.type==="throw"){const a=preSession.inspect().lastAbrupt;source=a===null?"":a.unitId;return {stage,harnessValidated:validated,status:classify(pre),source,host};}
  validated=true;load(testRealm,testSession,"attempt/reloaded");setup(testRealm);stage="test";
  const result=run('assert.throws(TypeError,function(){fault();});',testRealm,testSession,"attempt/test");
  const a=testSession.inspect().lastAbrupt;source=a===null?"":a.unitId;
  return {stage,harnessValidated:validated,status:classify(result),source,host};
 }catch(e){if(!(e instanceof Error))throw e;host=e;return {stage,harnessValidated:validated,status:"host",source,host};}
 finally {idle(preSession);idle(testSession);}
}
for(const mode of ["host","unsupported","resource"])for(const pre of [true,false]){
 const a=attempt(mode,pre);check(a.status===mode&&a.stage===(pre?"harness":"test")&&a.harnessValidated===!pre,"failure priority "+mode);
 if(mode==="host")check(a.host===hostSentinel,"exact escaping host identity");
}
console.log("PASS stage/failure priority controls=6");
