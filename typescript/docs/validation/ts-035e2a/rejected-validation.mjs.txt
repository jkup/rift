function files(directory,prefix='') {
  return readdirSync(directory,{withFileTypes:true}).flatMap(entry=>{
    const path=join(directory,entry.name),rel=join(prefix,entry.name);
    if(path===own)return [];
    return entry.isDirectory()?files(path,rel):[rel];
  });
}
