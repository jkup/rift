// Pure validation of retained status data; no process creation or build authority.
const resultKeys=['Outcome','StopReason','Errors','Events','Pid','NativeExitCode','Signal','CreatedSuspended','Assigned','Resumed','JobListConfigured','MembershipVerified','CleanupConfirmed','FinalActiveProcesses','ElapsedMs','Stdout','Stderr','ExternalEffects'];
const pipeKeys=['ObservedBytes','RetainedBytes','ObservedSha256','Error','Eof','Complete','Overflow'];
const wrapperKeys=['Outcome','Stage','Error','Exception','ScriptStackTrace','CleanupConfirmed','CompilerOutcome','ExternalEffects'];
const effects='not monitored outside enumerated roots; cooperative installed-tool and OS trust';
const keys=(value,names)=>{assert(value&&Object.getPrototypeOf(value)===Object.prototype);assert.deepEqual(Object.keys(value).sort(),[...names].sort());};
const uint=value=>Number.isSafeInteger(value)&&value>=0&&value<=0xffffffff;
const nullableString=value=>value===null||typeof value==='string';
export function inspectStatus(status) {
 if(status?.Outcome==='wrapper-fault'){
  keys(status,wrapperKeys);assert(['add-type','prepare-run','invoke'].includes(status.Stage));
  for(const name of ['Error','Exception'])assert.equal(typeof status[name],'string');assert(nullableString(status.ScriptStackTrace));
  assert.equal(status.CleanupConfirmed,false);assert.equal(status.CompilerOutcome,status.Stage==='invoke'?'unknown':'not-invoked');assert.equal(status.ExternalEffects,effects);
  return {kind:'wrapper-fault',compilerInvocations:status.Stage==='invoke'?null:0,cleanupReported:false};
 }
 keys(status,resultKeys);assert(['completed-observation','failed-observation-stop-review'].includes(status.Outcome));
 assert.equal(typeof status.StopReason,'string');for(const name of ['Errors','Events'])assert(Array.isArray(status[name])&&status[name].every(x=>typeof x==='string'));
 for(const name of ['Pid','NativeExitCode','FinalActiveProcesses'])assert(status[name]===null||uint(status[name]));
 assert.equal(status.Signal,null);assert(Number.isSafeInteger(status.ElapsedMs)&&status.ElapsedMs>=0);
 for(const name of ['CreatedSuspended','Assigned','Resumed','JobListConfigured','MembershipVerified','CleanupConfirmed'])assert.equal(typeof status[name],'boolean');
 assert.equal(status.ExternalEffects,effects);assert.equal(status.Assigned,status.CreatedSuspended);
 if(status.CreatedSuspended)assert(status.Pid>0);else assert.equal(status.Pid,null);
 if(status.MembershipVerified)assert(status.CreatedSuspended&&status.Assigned&&status.JobListConfigured);
 if(status.Resumed)assert(status.MembershipVerified);if(status.CleanupConfirmed)assert.equal(status.FinalActiveProcesses,0);
 for(const name of ['Stdout','Stderr']){const s=status[name];if(s===null)continue;keys(s,pipeKeys);
  for(const k of ['ObservedBytes','RetainedBytes'])assert(Number.isSafeInteger(s[k])&&s[k]>=0);
  assert(s.RetainedBytes<=8388608&&s.RetainedBytes<=s.ObservedBytes);assert(nullableString(s.Error));assert(s.ObservedSha256===null||(typeof s.ObservedSha256==='string'&&/^[a-f0-9]{64}$/.test(s.ObservedSha256)));
  for(const k of ['Eof','Complete','Overflow'])assert.equal(typeof s[k],'boolean');
 }
 return {kind:'supervisor-result',compilerInvocations:status.CreatedSuspended?1:0,cleanupReported:status.CleanupConfirmed};
}
export function validateCompletedStatus(status,stdout,stderr) {
 const observed=inspectStatus(status);assert.equal(observed.kind,'supervisor-result');
 assert.equal(status.Outcome,'completed-observation');assert.equal(status.StopReason,'root-exit');assert.deepEqual(status.Errors,[]);
 assert.deepEqual(status.Events,['job-kill-on-close','creation-time-job-list-configured','created-suspended-in-job','membership-verified','resumed','terminate-job']);
 for(const name of ['CreatedSuspended','Assigned','Resumed','JobListConfigured','MembershipVerified','CleanupConfirmed'])assert.equal(status[name],true);
 assert.equal(status.NativeExitCode,0);assert.equal(status.FinalActiveProcesses,0);
 for(const [name,bytes]of [['Stdout',stdout],['Stderr',stderr]]){const s=status[name];assert(s);assert(Buffer.isBuffer(bytes));
  assert.equal(s.Eof,true);assert.equal(s.Complete,true);assert.equal(s.Overflow,false);assert.equal(s.Error,null);
  assert.equal(s.ObservedBytes,bytes.length);assert.equal(s.RetainedBytes,bytes.length);assert.equal(s.ObservedSha256,hash(bytes));
 }
 return observed;
}
