param([Parameter(Mandatory=$true)][string]$ConfigPath)
$ErrorActionPreference = 'Stop'
$expectedConfigSha256 = 'c3b00705ae91fd9e376f6f8f40c54424f9ca82409d1495efcc3715856c294246'
$configBytes = [IO.File]::ReadAllBytes($ConfigPath)
if ($configBytes.Length -gt 65536) { throw 'config cap' }
$configHash = [Convert]::ToHexString([Security.Cryptography.SHA256]::HashData($configBytes)).ToLowerInvariant()
if ($configHash -ne $expectedConfigSha256) { throw 'config identity' }
$config = Get-Content -LiteralPath $ConfigPath -Raw | ConvertFrom-Json
$statusPath = Join-Path $config.root 'supervisor-status.json'
$result = $null
$stage = 'add-type'
try {
    # Future execution dependency only. No Add-Type during design validation.
    Add-Type -Path (Join-Path $config.root 'supervisor.cs')
    $stage = 'prepare-run'
    $environmentPairs = @($config.childEnvironment.PSObject.Properties | ForEach-Object { $_.Name + '=' + [string]$_.Value })
    $stage = 'invoke'
    $result = [ProbeSupervisor]::Run(
        ([string]$config.executable),
        [string[]]$config.argv, [string[]]$environmentPairs, [string]$config.root,
        [int]$config.timeoutMs, [int]$config.streamCapBytes, [int]$config.cleanupMs)
} catch {
    $result = [ordered]@{Outcome='wrapper-fault';Stage=$stage;Error=$_.ToString();Exception=$_.Exception.ToString();ScriptStackTrace=$_.ScriptStackTrace;CleanupConfirmed=$false;CompilerOutcome=$(if($stage -eq 'invoke'){'unknown'}else{'not-invoked'});ExternalEffects='not monitored outside enumerated roots; cooperative installed-tool and OS trust'}
}
try {
    $bytes = [Text.Encoding]::UTF8.GetBytes(($result | ConvertTo-Json -Depth 20) + "`n")
    $file = [IO.File]::Open($statusPath, [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::Read)
    try { $file.Write($bytes, 0, $bytes.Length); $file.Flush($true) } finally { $file.Dispose() }
} catch {
    [Console]::Error.WriteLine($_.ToString())
    exit 74
}
if ($result.Outcome -eq 'completed-observation' -and $result.CleanupConfirmed) { exit 0 }
exit 70
