function supervise(p,root) {
 return new Promise(resolveResult=>{
  const record={exe:p.supervisor,argv:p.supervisorArgv,cwd:root,environment:p.environment,spawned:false,pid:null,error:null,exit:null,signal:null,watchdog:false,killRequested:false,killReturned:null,closeObserved:false,streams:{stdout:{bytes:0,overflow:false},stderr:{bytes:0,overflow:false}}};
  const descriptors={},errors=[];let child,ended=false,watchdog,closeDeadline;
  const finish=()=>{if(ended)return;ended=true;clearTimeout(watchdog);clearTimeout(closeDeadline);
   for(const fd of Object.values(descriptors)){try{fsyncSync(fd);}catch(e){errors.push(String(e));}finally{try{closeSync(fd);}catch(e){errors.push(String(e));}}}
   record.ioErrors=errors;resolveResult(record);
  };
  const stop=()=>{if(record.killRequested||ended)return;record.killRequested=true;
   try{record.killReturned=child.kill();}catch(e){errors.push(String(e));}
   closeDeadline=setTimeout(()=>{errors.push('supervisor close unconfirmed; stop for manual inspection, no retry');finish();},10000);
  };
  try{for(const key of ['stdout','stderr'])descriptors[key]=openSync(join(root,'supervisor.'+key+'.bin'),'wx');}
  catch(e){record.error={code:e.code??null,message:e.message};finish();return;}
  try{child=spawn(p.supervisor,p.supervisorArgv,{cwd:root,env:p.environment,windowsHide:true,stdio:['ignore','pipe','pipe']});}
  catch(e){record.error={code:e.code??null,message:e.message};finish();return;}
  child.on('spawn',()=>{record.spawned=true;record.pid=child.pid;});
  child.on('error',e=>{if(ended)return;record.error={code:e.code??null,message:e.message};if(record.spawned)stop();});
  for(const key of ['stdout','stderr']){
   child[key].on('error',e=>{if(ended)return;errors.push(key+': '+String(e));stop();});
   child[key].on('data',chunk=>{
    if(ended)return;const s=record.streams[key],previous=s.bytes;s.bytes+=chunk.length;
    try{const limit=Math.max(0,Math.min(chunk.length,1048577-previous));let offset=0;while(offset<limit){const n=writeSync(descriptors[key],chunk,offset,limit-offset);if(n<=0)throw Error('supervisor-stream-zero-write');offset+=n;}}
    catch(e){errors.push(String(e));stop();}
    if(s.bytes>1048576){s.overflow=true;stop();}
   });
  }
  child.on('exit',(code,signal)=>{if(ended)return;record.exit=code;record.signal=signal;});
  child.on('close',(code,signal)=>{if(ended)return;record.exit=code;record.signal=signal;record.closeObserved=true;finish();});
  watchdog=setTimeout(()=>{record.watchdog=true;stop();},p.supervisorWatchdogMs);
 });
}
