function obj(object: GuestObject): Value { return { kind: "object", object }; }
function str(value: string): Value { return { kind: "string", value }; }

function data(object: GuestObject, key: string, value: Value): void { object.bootstrapDefine(key, { value, writable: true, configurable: true }); }

function own(object: GuestObject, key: string): Value {
    const descriptor = object.getOwnProperty(key);
    if (descriptor === undefined || descriptor.kind !== "data") throw new Error("missing data " + key);
    return descriptor.value;
}
function object(value: Value): GuestObject { if (value.kind !== "object") throw new Error("object"); return value.object; }
function prepare(realm: Realm, text: string, id: string): ExecutableScript {
    const prepared = prepareBoundedScript({ id, text }, realm);
    if (prepared.kind !== "ready") throw new Error("prepare " + id);
    return prepared.record;
}
function idle(session: ScriptSession): void {
    const state = session.inspect();
    check(state.active === null && state.calls === 0 && state.depth === 0, "all-exit cleanup");
}
function run(session: ScriptSession, realm: Realm, text: string, id: string): ValueCompletion {
    try { return evaluateScript(session, prepare(realm, text, id)).completion; } finally { idle(session); }
}
function observe(session: ScriptSession, realm: Realm, text: string, id: string): ObservedScriptEvaluation {
    try { return evaluateScriptObserved(session, prepare(realm, text, id)); } finally { idle(session); }
}
function abrupt(result: ObservedScriptEvaluation): ThrowCompletion {
    const completion = result.evaluation.completion;
    if (completion.type !== "throw") throw new Error("original throw");
    return completion;
}
function available(result: ObservedScriptEvaluation, realm: Realm, name: string): void {
    const inspection = result.inspection;
    check(inspection !== null && inspection.kind === "available" && inspection.realm === realm && inspection.name === name, "available " + name);
}
function retained(session: ScriptSession, result: ObservedScriptEvaluation): void {
    const source = result.source, history = session.inspect().lastAbrupt;
    check(source !== null && history !== null && source.record === history.record && source.realm === history.realm &&
        source.unitId === history.unitId && source.loc.line === history.loc.line && source.loc.column === history.loc.column, "original history");
    if (source === null || history === null) throw new Error("source");
    source.loc.column = 999;
    check(session.inspect().lastAbrupt?.loc.column === history.loc.column, "detached result location");
    source.loc.column = history.loc.column;
}
function accessor(target: GuestObject, key: string, realm: Realm, callback: Builtin): void {
    target.bootstrapDefine(key, { get: obj(realm.createBuiltin(key, 0, callback)), configurable: true });
