import assert from 'node:assert/strict';
import * as fs from 'node:fs';
import {join} from 'node:path';
import {validateAliases,classify,validateExecution} from './policy.mjs';
import {parseAliasList} from './runtime-support.mjs';
const here=import.meta.dirname,json=n=>JSON.parse(fs.readFileSync(join(here,n))),alias=json('alias-policy.json'),outputs=json('output-policy.json'),dirs=json('staging-policy.json').privateDirectories,observations=[];
const check=(id,fn,expected)=>{let actual;try{actual=fn();}catch(e){assert.equal(e.code,'ERR_ASSERTION',id+' intended rejection');actual={kind:'rejected',code:e.code};}assert.deepEqual(actual,expected,id);observations.push({id,status:'passed'});};
try{
 for(const f of json('alias-literals.json'))check('alias/'+f.id,()=>validateAliases(f.input,alias),f.expected);
 for(const f of json('output-literals.json'))check('output/'+f.id,()=>classify(f.rows,outputs,dirs),f.expected);
 for(const f of json('status-literals.json'))check('status/'+f.id,()=>validateExecution(f.input.outer,f.input.status,{stdout:Buffer.from(f.input.stdoutHex,'hex'),stderr:Buffer.from(f.input.stderrHex,'hex')},f.input.context),f.expected);
 const audit=json('alias-audit.json');for(let i=0;i<6;i++)for(const phase of ['before','after'])check('parse-retained-utility/'+i+'/'+phase,()=>parseAliasList(fs.readFileSync(join(here,`aliases-${i}-${phase}.stdout.bin`)),audit.files[i].named),audit.files[i].names);
 const source=fs.readFileSync(join(here,'policy.mjs'),'utf8');
 const mutations=[
  ['alias-policy-drift',"assert.deepEqual(a.names,e.names,'alias names/order');","",()=>{const f=json('alias-literals.json').find(f=>f.id==='extra-link-0');return {type:'alias',fixture:f};}],
  ['unexpected-output-hidden',"else failures.push('unexpected-file:'+r.path);","else {}",()=>({type:'output',fixture:json('output-literals.json').find(f=>f.id==='unexpected-file')})],
  ['cleanup-ignored',"'MembershipVerified','CleanupConfirmed'","'MembershipVerified'",()=>({type:'status',fixture:json('status-literals.json').find(f=>f.id==='cleanup-unconfirmed')})]
 ];
 // Alias-name removal alone still hits count/identity guards; do not claim a killed mutant from setup failure.
 for(const [id,old,next,pick] of mutations.slice(1)){assert.equal(source.split(old).length-1,1);const m=await import('data:text/javascript;base64,'+Buffer.from(source.replace(old,next)).toString('base64')),f=pick().fixture;let actual;if(pick().type==='output')actual=m.classify(f.rows,outputs,dirs);else actual=m.validateExecution(f.input.outer,f.input.status,{stdout:Buffer.from(f.input.stdoutHex,'hex'),stderr:Buffer.from(f.input.stderrHex,'hex')},f.input.context);let e;try{assert.deepEqual(actual,f.expected);}catch(err){e=err;}assert.equal(e?.code,'ERR_ASSERTION');observations.push({id:'mutant/'+id,status:'passed'});}
 fs.writeFileSync(process.argv[2],JSON.stringify({status:'passed',observations,aliasLiterals:46,outputLiterals:13,statusLiterals:19,retainedUtilityParses:12,mutantsKilled:2,compilerInvocations:0,addTypeInvocations:0,newUtilityInvocations:0},null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({status:'passed',controls:observations.length}));
}catch(e){fs.writeFileSync(process.argv[2],JSON.stringify({status:'failed',observations,error:{name:e.name,message:e.message,actual:e.actual,expected:e.expected}},null,2)+'\n',{flag:'wx'});throw e;}
